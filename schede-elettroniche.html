void loop() {

    // Touch — debounce 500 ms
    auto t = MSDial.Touch.getDetail();
    if (t.isPressed()) {
      if ((millis() - clickPrecedente) >= 500) {
        clickPrecedente = millis();

        switch (paginaAttuale) {

          case paginaImpostaTimer:
            setMinuto = (t.x < 120);
            MSDial.Speaker.tone(3000, 100);
            break;

          case paginaImpostaMaster:
            if (t.y > 120)
              setDecine = (t.x < 120);
            else
              PannelloAcceso = !PannelloAcceso;
            MSDial.Speaker.tone(3000, 100);
            break;

          default: break;
        }
      }
    }
  }
